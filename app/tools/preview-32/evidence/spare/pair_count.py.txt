import subprocess, sys, time, re, xml.etree.ElementTree as ET
SER='ZY22KW6P9J'
def sh(*a): return subprocess.run(['adb','-s',SER,'shell',*a],capture_output=True,text=True,timeout=60).stdout
def dump():
    for _ in range(3):
        out=sh('uiautomator dump /sdcard/ui.xml >/dev/null && cat /sdcard/ui.xml && rm /sdcard/ui.xml')
        if '<hierarchy' in out: return ET.fromstring(out[out.index('<?xml') if '<?xml' in out else out.index('<hierarchy'):])
        time.sleep(1)
    raise SystemExit('FAIL: no dump')
def nodes(root): return list(root.iter('node'))
def texts(root): return [n.get('text') or n.get('content-desc') or '' for n in nodes(root)]
def center(n):
    x1,y1,x2,y2=map(int,re.findall(r'\d+',n.get('bounds'))); return (x1+x2)//2,(y1+y2)//2
def tap(label, tries=10):
    for _ in range(tries):
        r=dump()
        for n in nodes(r):
            if (n.get('text') or n.get('content-desc'))==label:
                x,y=center(n); sh('input','tap',str(x),str(y)); print('PASS: tap',label); time.sleep(1.5); return
        sh('input','swipe','540','1800','540','900','300'); time.sleep(1)
    raise SystemExit('FAIL: not found '+label)
